-- Database: tiendatecnologica
create table cliente(
id		bigint generated by default as identity primary key,
nombre		varchar(60) not null,
apellido	varchar(60) not null,
-- Fono debe ser varchar
telefono	varchar(20) not null,
email		varchar(150) not null unique,
-- Nuevos campos para practicar fechas
fecha_nacimiento date not null,
fecha_alta       timestamp not null default current_timestamp,
-- Restricción: La fecha de nacimiento debe ser anterior a hoy
constraint ck_cliente_fecha_nacimiento check(fecha_nacimiento < current_date)
);

create table proveedor(
id		bigint generated by default as identity primary key,
nombre		varchar(60) not null,
-- Apellido sin not full para empresas
apellido	varchar(60),
-- Fono debe ser varchar
telefono	varchar(20) not null,
email		varchar(150) not null unique
);

create table categoria (
id		bigint generated by default as identity primary key,
nombre	varchar(100) not null unique
);

create table producto (
id		bigint generated by default as identity primary key,
nombre		varchar(150) not null,
cantidad	integer not null,
precio		numeric(10,2) not null,
categoria_id	bigint not null,
proveedor_id	bigint not null,
constraint fk_producto_proveedor foreign key(proveedor_id) references proveedor(id),
constraint fk_producto_categoria foreign key(categoria_id) references categoria(id),

constraint ck_producto_cantidad check(cantidad >= 0),
constraint ck_producto_precio check(precio >= 0)
);

create table imagen (
    id           bigint generated by default as identity primary key,
    producto_id  bigint not null,
    imagen_url   varchar(500) not null, -- Guarda la subcarpeta y el nombre (ej: 'producto-1/p1ima1.jpg')
    constraint fk_imagen_producto foreign key (producto_id) 
        references producto(id) on delete cascade
);

-- Consultar productos con categoria y proveedor
SELECT
    p.id,
    p.nombre AS producto,
    p.cantidad,
    p.precio,
    c.nombre AS categoria,
    pr.nombre || ' ' || pr.apellido AS proveedor
FROM producto p
INNER JOIN categoria c ON p.categoria_id = c.id
INNER JOIN proveedor pr ON p.proveedor_id = pr.id
ORDER BY p.id;
